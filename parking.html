第二部分：Google Sheet 雲端同步後端（Code.gs 選配） 若希望 4 位老師手機即時同步進場數量，並將新登記的校長或嘉賓車牌自動寫入 Google Sheet 兩個分頁（永久車牌總庫 與 當日進場記錄），請將以下程式碼貼入 Google Sheet 的 Apps Script (Code.gs) 中： const MASTER_SHEET_NAME = "永久車牌總庫"; const CHECKIN_SHEET_NAME = "當日進場記錄"; function doGet(e) { if (e && e.parameter && e.parameter.api === "1") { const state = getParkingState(null); return ContentService .createTextOutput(JSON.stringify(state)) .setMimeType(ContentService.MimeType.JSON); } return HtmlService.createHtmlOutputFromFile("Index") .setTitle("學校車牌掃描及極速分流系統") .addMetaTag("viewport", "width=device-width, initial-scale=1, maximum-scale=1, user-scalable=no") .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function doPost(e) { const result = handleClientAction(e.postData.contents); return ContentService .createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function ensureSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let masterSheet = ss.getSheetByName(MASTER_SHEET_NAME); if (!masterSheet) { masterSheet = ss.insertSheet(MASTER_SHEET_NAME); masterSheet.appendRow(["ID", "機構/學校", "姓名", "原始車牌", "標準化車牌", "職銜", "預設分流", "是否歷史新增", "備註"]); } let checkInSheet = ss.getSheetByName(CHECKIN_SHEET_NAME); if (!checkInSheet) { checkInSheet = ss.insertSheet(CHECKIN_SHEET_NAME); checkInSheet.appendRow(["ID", "車牌", "車主", "實際泊位", "進場時間", "放行老師", "時間戳記"]); } return { masterSheet, checkInSheet }; } function getParkingState(initialMasterJson) { const lock = LockService.getScriptLock(); lock.tryLock(5000); try { const { masterSheet, checkInSheet } = ensureSheets(); // 若試算表尚無預載名單，自動寫入前端傳來的 27 架初始車輛 if (masterSheet.getLastRow() <= 1 && initialMasterJson) { const initArr = JSON.parse(initialMasterJson); const rows = initArr.map(v => [ v.id, v.org, v.name, v.rawPlate, v.cleanPlate, v.role, v.zone, v.isHistory ? "YES" : "NO", v.note || "" ]); if (rows.length > 0) { masterSheet.getRange(2, 1, rows.length, 9).setValues(rows); } } const masterList = []; if (masterSheet.getLastRow() > 1) { const mData = masterSheet.getRange(2, 1, masterSheet.getLastRow() - 1, 9).getValues(); mData.forEach(r => { if (!r[0]) return; masterList.push({ id: String(r[0]), org: String(r[1]), name: String(r[2]), rawPlate: String(r[3]), cleanPlate: String(r[4]), role: String(r[5]), zone: String(r[6]), isHistory: String(r[7]) === "YES", note: String(r[8] || "") }); }); } const checkInMap = {}; if (checkInSheet.getLastRow() > 1) { const cData = checkInSheet.getRange(2, 1, checkInSheet.getLastRow() - 1, 6).getValues(); cData.forEach(r => { if (!r[0]) return; checkInMap[String(r[0])] = { id: String(r[0]), plate: String(r[1]), owner: String(r[2]), zone: String(r[3]), time: String(r[4]), by: String(r[5]) }; }); } return { ok: true, masterList, checkInMap }; } finally { lock.releaseLock(); } } function handleClientAction(payloadStr) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const payload = JSON.parse(payloadStr); const { masterSheet, checkInSheet } = ensureSheets(); if (payload.action === "resetCheckIns") { if (checkInSheet.getLastRow() > 1) { checkInSheet.deleteRows(2, checkInSheet.getLastRow() - 1); } return getParkingState(null); } if (payload.vehicle) { const v = payload.vehicle; let foundRow = -1; if (masterSheet.getLastRow() > 1) { const ids = masterSheet.getRange(2, 1, masterSheet.getLastRow() - 1, 1).getValues(); for (let i = 0; i < ids.length; i++) { if (String(ids[i][0]) === String(v.id)) { foundRow = i + 2; break; } } } const rowValues = [ v.id, v.org, v.name, v.rawPlate, v.cleanPlate, v.role, v.zone, v.isHistory ? "YES" : "NO", v.note || "" ]; if (foundRow > 0) { masterSheet.getRange(foundRow, 1, 1, 9).setValues([rowValues]); } else { masterSheet.appendRow(rowValues); } } if (payload.checkIn) { const c = payload.checkIn; checkInSheet.appendRow([c.id, c.plate, c.owner, c.zone, c.time, c.by, new Date()]); } return getParkingState(null); } finally { lock.releaseLock(); } }
